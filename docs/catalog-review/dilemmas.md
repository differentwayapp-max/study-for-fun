  - 2 = אחרי השלמת הכיסוי (sff-v119)
  - 3 = אחרי תיקוני התמונות (sff-v126)
  - 4 = גרסת ביניים של תיקוני סבב 4
  - 5 = אחרי סריקת משחקי ההתאמה (sff-v128)
